Aqui está o código completo do **`app.py`** refatorado e reorganizado exatamente de acordo com a arquitetura de 3 blocos do **DMAIC Ágil** apresentada no diagrama:

1. **Bloco 1 — Definição Ágil (Sprint 1: 2 semanas):** Time, VOC & CTQs, Indicadores (Y), SIPOC, Matriz Dentro/Fora e Project Charter.
2. **Bloco 2 — Medição Ágil (Sprint 2: 2 a 3 semanas):** Validação MSA (Gage R&R), Estatística Descritiva, Gráfico de Pareto, Ishikawa 6M, Matriz Causa & Efeito, Matrizes GUT / Esforço x Impacto e Desdobramento da Meta em $X$'s vitais.
3. **Bloco 3 — Ciclo Iterativo A-I-C (Sprints de 2 semanas por $X$ vital - *One X Flow*):**
* **Análise Ágil:** Comprovação dos $X$'s (ANOVA, Teste t, Regressão, FMEA).
* **Melhoria Ágil:** Brainstorming, Diagrama de Árvore, Matriz de Priorização de Soluções e 5W2H.
* **Controle Ágil:** Cartas de Controle I-MR (Western Electric), Capabilidade Antes/Depois e Padronização (POP, Poka Yoke, OCAP e Plano de Controle).

def calculate_pareto(df: pd.DataFrame, category_col: str, count_col: Optional[str] = None):
    """Gera tabela e gráfico de Pareto 80/20 com duplo eixo."""
    if count_col and count_col in df.columns:
        grouped = df.groupby(category_col)[count_col].sum().reset_index()
    else:
        grouped = df[category_col].value_counts().reset_index()
        grouped.columns = [category_col, 'count']
        count_col = 'count'

    grouped = grouped.sort_values(by=count_col, ascending=False).reset_index(drop=True)
    grouped['cumulative_sum'] = grouped[count_col].cumsum()
    grouped['cumulative_perc'] = (grouped['cumulative_sum'] / grouped[count_col].sum()) * 100

    fig = go.Figure()
    fig.add_trace(go.Bar(
        x=grouped[category_col], 
        y=grouped[count_col], 
        name='Frequência',
        marker_color='#1f77b4'
    ))
    fig.add_trace(go.Scatter(
        x=grouped[category_col], 
        y=grouped['cumulative_perc'], 
        name='% Acumulado', 
        yaxis='y2', 
        mode='lines+markers',
        marker_color='#d62728',
        line=dict(width=2)
    ))
    fig.add_hline(y=80, line_dash="dash", line_color="orange", yref="y2", annotation_text="Corte 80%")

    fig.update_layout(
        title="Gráfico de Pareto (Itens Vitais)",
        yaxis=dict(title="Ocorrências / Volume"),
        yaxis2=dict(
            title="% Acumulado", 
            overlaying='y', 
            side='right', 
            range=[0, 105],
            ticksuffix="%"
        ),
        legend=dict(x=0.8, y=1.15, orientation="h"),
        margin=dict(l=40, r=40, t=50, b=40)
    )
    return grouped, fig


def calculate_imr_chart(df: pd.DataFrame, value_col: str):
    """Calcula Cartas I-MR com regras de Nelson / Western Electric aplicadas."""
    series = df[value_col].dropna().reset_index(drop=True)
    n = len(series)
    if n < 3:
        raise ValueError("Mínimo de 3 observações necessárias para cálculo da carta I-MR.")

    # Carta I
    x_bar = series.mean()
    mr = series.diff().abs().dropna().reset_index(drop=True)
    mr_bar = mr.mean()
    sigma_est = mr_bar / 1.128  # d2 para n=2
    
    ucl_i = x_bar + 3 * sigma_est
    lcl_i = max(0.0, x_bar - 3 * sigma_est)
    
    violations_i = set()
    for i, val in enumerate(series):
        if val > ucl_i or val < lcl_i:
            violations_i.add(i)
            
    # Regra de 9 pontos consecutivos no mesmo lado da média
    for i in range(len(series) - 8):
        window = series[i:i+9]
        if (window > x_bar).all() or (window < x_bar).all():
            for idx in range(i, i+9):
                violations_i.add(idx)

    # Carta MR
    ucl_mr = 3.267 * mr_bar
    lcl_mr = 0.0
    violations_mr = set([i + 1 for i, val in enumerate(mr) if val > ucl_mr])

    fig = make_subplots(
        rows=2, cols=1, shared_xaxes=True, vertical_spacing=0.12,
        subplot_titles=("Carta Individual (I)", "Carta de Amplitude Móvel (MR)")
    )

    indices = list(range(1, n + 1))
    colors_i = ['#d62728' if i in violations_i else '#1f77b4' for i in range(n)]

    fig.add_trace(
        go.Scatter(x=indices, y=series, mode='lines+markers', name='Individual',
                   marker=dict(color=colors_i, size=8), line=dict(color='#1f77b4', width=1.5)),
        row=1, col=1
    )
    fig.add_hline(y=x_bar, line_dash="dash", line_color="blue", annotation_text=f"Média: {x_bar:.2f}", row=1, col=1)
    fig.add_hline(y=ucl_i, line_dash="dot", line_color="red", annotation_text=f"LSC: {ucl_i:.2f}", row=1, col=1)
    fig.add_hline(y=lcl_i, line_dash="dot", line_color="red", annotation_text=f"LIC: {lcl_i:.2f}", row=1, col=1)

    mr_indices = list(range(2, n + 1))
    colors_mr = ['#d62728' if (i + 1) in violations_mr else '#2ca02c' for i in range(len(mr))]
    
    fig.add_trace(
        go.Scatter(x=mr_indices, y=mr, mode='lines+markers', name='Amplitude Móvel',
                   marker=dict(color=colors_mr, size=7), line=dict(color='#2ca02c', width=1.5)),
        row=2, col=1
    )
    fig.add_hline(y=mr_bar, line_dash="dash", line_color="blue", annotation_text=f"MR Média: {mr_bar:.2f}", row=2, col=1)
    fig.add_hline(y=ucl_mr, line_dash="dot", line_color="red", annotation_text=f"LSC: {ucl_mr:.2f}", row=2, col=1)
    fig.add_hline(y=lcl_mr, line_dash="dot", line_color="red", annotation_text=f"LIC: {lcl_mr:.2f}", row=2, col=1)

    fig.update_layout(height=600, showlegend=False, margin=dict(l=40, r=40, t=50, b=40))
    stats_summary = {
        "x_bar": x_bar, "sigma_est": sigma_est, "ucl_i": ucl_i, "lcl_i": lcl_i,
        "mr_bar": mr_bar, "ucl_mr": ucl_mr, "violations_i_count": len(violations_i)
    }
    return fig, stats_summary
