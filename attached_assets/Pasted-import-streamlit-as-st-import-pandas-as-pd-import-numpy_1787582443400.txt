import streamlit as st
import pandas as pd
import numpy as np
import plotly.graph_objects as go
from langchain_google_genai import ChatGoogleGenerativeAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
import os

# ==========================================
# 1. FUNÇÕES ESTATÍSTICAS DETERMINÍSTICAS
# ==========================================
def plot_time_series(df: pd.DataFrame, time_col: str, value_col: str):
    """Gera gráfico sequencial de séries temporais com linha de média."""
    mean_val = df[value_col].mean()
    
    fig = go.Figure()
    fig.add_trace(go.Scatter(
        x=df[time_col], 
        y=df[value_col], 
        mode='lines+markers',
        name='Série Temporal',
        line=dict(color='#1f77b4', width=2),
        marker=dict(size=7)
    ))
    fig.add_hline(
        y=mean_val, 
        line_dash="dash", 
        line_color="red", 
        annotation_text=f"Média: {mean_val:.2f}"
    )
    fig.update_layout(
        title=f"Gráfico Sequencial: {value_col} ao longo do tempo",
        xaxis_title=time_col,
        yaxis_title=value_col,
        margin=dict(l=40, r=40, t=40, b=40)
    )
    return fig

def plot_custom_boxplot(df: pd.DataFrame, value_col: str):
    """Calcula quartis exatos e renderiza Boxplot anotado."""
    series = df[value_col].dropna()
    
    q1 = float(np.percentile(series, 25))
    median = float(np.percentile(series, 50))
    q3 = float(np.percentile(series, 75))
    min_val = float(series.min())
    max_val = float(series.max())
    iqr = q3 - q1

    fig = go.Figure()
    fig.add_trace(go.Box(
        y=series, 
        name=value_col,
        boxpoints='outliers',
        marker_color='#2ca02c',
        line=dict(width=1.5)
    ))
    
    # Anotações gráficas de Q1 e Q3
    fig.add_annotation(x=0, y=q1, text=f"1º Quartil (Q1): {q1:.2f}", showarrow=True, arrowhead=2, ax=60)
    fig.add_annotation(x=0, y=q3, text=f"3º Quartil (Q3): {q3:.2f}", showarrow=True, arrowhead=2, ax=60)
    
    fig.update_layout(
        title=f"Boxplot com Destaque de Quartis ({value_col})",
        yaxis_title=value_col,
        margin=dict(l=40, r=40, t=40, b=40)
    )

    stats = {
        "min": min_val,
        "q1": q1,
        "median": median,
        "q3": q3,
        "max": max_val,
        "iqr": iqr
    }
    return fig, stats

# ==========================================
# 2. INTERPRETAÇÃO COGNITIVA COM GEMINI
# ==========================================
def analyze_time_series_patterns(df: pd.DataFrame, time_col: str, value_col: str):
    """Aciona o Gemini para identificar tendências, sazonalidade e desvios."""
    data_summary = df[[time_col, value_col]].to_dict(orient="records")
    
    llm = ChatGoogleGenerativeAI(
        model="gemini-1.5-pro",
        temperature=0.2,
        google_api_key=os.getenv("GOOGLE_API_KEY")
    )
    
    prompt = ChatPromptTemplate.from_messages([
        ("system", """Você é um Master Black Belt em Lean Six Sigma.
Analise os dados de série temporal fornecidos e elabore um diagnóstico técnico:
1. Avalie se existe algum padrão de comportamento relevante (tendência de alta/baixa, ciclos, saltos ou estabilidade).
2. Explique detalhadamente o que está ocorrendo com o resultado ao longo dos períodos/meses."""),
        ("user", "Dados da série temporal:\n{data}")
    ])
    
    chain = prompt | llm | StrOutputParser()
    return chain.invoke({"data": str(data_summary)})

# 3. INTERFACE STREAMLIT
# ==========================================
def render_exploratory_analysis_tab():
    st.subheader("📊 Análise Sequencial (Séries Temporais) & Boxplot")
    
    # Dataset demonstrativo com datas e valores
    sample_df = pd.DataFrame({
        "Mes": ["Jan/24", "Fev/24", "Mar/24", "Abr/24", "Mai/24", "Jun/24", "Jul/24", "Ago/24", "Set/24", "Out/24", "Nov/24", "Dez/24"],
        "Resultado_Indicador": [45.2, 44.8, 46.1, 48.3, 51.0, 52.4, 55.1, 54.8, 58.2, 60.1, 62.4, 65.0]
    })
    
    uploaded_file = st.file_uploader("Subir CSV/Excel para Análise Exploratória", type=["csv", "xlsx"], key="exploratory_upload")
    df = sample_df if not uploaded_file else (pd.read_csv(uploaded_file) if uploaded_file.name.endswith(".csv") else pd.read_excel(uploaded_file))
    
    c1, c2 = st.columns(2)
    with c1:
        time_column = st.selectbox("Coluna Temporal (Meses/Período)", options=df.columns, index=0)
    with c2:
        numeric_cols = [c for c in df.columns if pd.api.types.is_numeric_dtype(df[c])]
        val_column = st.selectbox("Coluna do Resultado / Indicador", options=numeric_cols, index=len(numeric_cols)-1)
    
    if st.button("Executar Diagnóstico Gráfico e Estatístico", type="primary"):
        # 1. Gráfico de Séries Temporais e Boxplot
        col_ts, col_box = st.columns([1, 1])
        
        with col_ts:
            fig_ts = plot_time_series(df, time_column, val_column)
            st.plotly_chart(fig_ts, use_container_width=True)
            
        with col_box:
            fig_box, stats = plot_custom_boxplot(df, val_column)
            st.plotly_chart(fig_box, use_container_width=True)
        
        # 2. Respostas Didáticas e Interpretação dos Quartis
        st.markdown("---")
        st.markdown("### 📌 Interpretação Conceitual da Distribuição")
        
        k1, k2, k3, k4, k5 = st.columns(5)
        k1.metric("Mínimo", f"{stats['min']:.2f}")
        k2.metric("1º Quartil (Q1 - 25%)", f"{stats['q1']:.2f}")
        k3.metric("Mediana (Q2 - 50%)", f"{stats['median']:.2f}")
        k4.metric("3º Quartil (Q3 - 75%)", f"{stats['q3']:.2f}")
        k5.metric("Máximo", f"{stats['max']:.2f}")
        
        col_q1, col_q2 = st.columns(2)
        with col_q1:
            st.info(
                f"**A região entre o Valor Mínimo ({stats['min']:.2f}) e o 1º Quartil ({stats['q1']:.2f}):**\n\n"
                "Representa exatamente os **25% menores valores** (primeiro quarto inferior) de todo o conjunto de dados."
            )
        with col_q2:
            st.info(
                f"**A região entre o 3º Quartil ({stats['q3']:.2f}) e o Valor Máximo ({stats['max']:.2f}):**\n\n"
                "Representa exatamente os **25% maiores valores** (primeiro quarto superior) de todo o conjunto de dados."
            )

        # 3. Diagnóstico do Gemini sobre o comportamento temporal
        st.markdown("### 🤖 Diagnóstico do Comportamento da Série Temporal (Gemini)")
        with st.spinner("Analisando padrões e tendências com o Gemini..."):
            try:
                ai_analysis = analyze_time_series_patterns(df, time_column, val_column)
                st.markdown(ai_analysis)
            except Exception as e:
                st.warning(f"Não foi possível obter a análise textual: {str(e)}")
